# Block Build Reference

Canonical bodies for a Gutenberg block generated by `/wp-pro-max:plugin add block <name>`.
All templates are embedded in this reference. The scaffold script extracts them by
marker, tokenizes them, and writes the block files. No separate `templates/` tree.

## File layout

```
blocks/<name>/
  block.json
  src/
    index.js
    edit.js
    style.scss
    editor.scss
package.json   (first block only)
src/Blocks/<Name>Block.php
```

The compiled `blocks/<name>/build/` directory is git-ignored but is
**runtime-required**; the `package` step force-includes it in the `.zip`.

## `src/Blocks/<Name>Block.php`

<!-- file: Blocks/HeroBlock.php -->
```php
<?php
/**
 * Block: __FEATURE_NAME__.
 *
 * @package __NAMESPACE__
 */

namespace __NAMESPACE__\Blocks;

defined( 'ABSPATH' ) || exit;

/**
 * __FEATURE_NAME__ dynamic block.
 */
class __FEATURE_CLASS__ {

	/**
	 * Block name.
	 *
	 * @var string
	 */
	private $name = '__SLUG__/__FEATURE_SLUG_KEBAB__';

	/**
	 * Block build directory.
	 *
	 * @var string
	 */
	private $build_dir = '';

	/**
	 * Register hooks.
	 */
	public function register() {
		add_action( 'init', array( $this, 'register_block' ) );
	}

	/**
	 * Register the block.
	 */
	public function register_block() {
		$this->build_dir = plugin_dir_path( __FILE__ ) . '../../blocks/__FEATURE_SLUG__/build';
		if ( ! is_dir( $this->build_dir ) ) {
			return;
		}
		register_block_type(
			$this->build_dir,
			array(
				'render_callback' => array( $this, 'render' ),
			)
		);
	}

	/**
	 * Render callback.
	 *
	 * @param array  $attributes Block attributes.
	 * @param string $content    Inner blocks content.
	 * @return string
	 */
	public function render( $attributes, $content ) { // phpcs:ignore Generic.CodeAnalysis.UnusedFunctionParameter.FoundAfterLastUsed
		$title = isset( $attributes['title'] ) ? sanitize_text_field( $attributes['title'] ) : '';

		ob_start();
		?>
		<div <?php echo wp_kses_data( get_block_wrapper_attributes() ); ?>>
			<h2><?php echo esc_html( $title ? $title : __( '__FEATURE_NAME__', '__TEXTDOMAIN__' ) ); ?></h2>
		</div>
		<?php
		return ob_get_clean();
	}
}
```

## `blocks/<name>/src/index.js`

<!-- file: blocks/hero/src/index.js -->
```js
import { registerBlockType } from '@wordpress/blocks';
import metadata from '../block.json';
import Edit from './edit';
import './style.scss';
import './editor.scss';

registerBlockType( metadata.name, {
	edit: Edit,
	save: () => null,
} );
```

## `blocks/<name>/src/edit.js`

<!-- file: blocks/hero/src/edit.js -->
```js
import { useBlockProps } from '@wordpress/block-editor';

export default function Edit( { attributes, setAttributes } ) {
	const blockProps = useBlockProps();

	return (
		<div { ...blockProps }>
			<input
				type="text"
				value={ attributes.title }
				onChange={ ( event ) => setAttributes( { title: event.target.value } ) }
			/>
		</div>
	);
}
```

## `package.json`

<!-- file: package.json -->
```json
{
  "name": "__SLUG__-blocks",
  "version": "__VERSION__",
  "private": true,
  "scripts": {
    "build": "for block in blocks/*/; do [ -f \"$block/block.json\" ] || continue; wp-scripts build --webpack-src-dir=\"$block/src\" --output-path=\"$block/build\" && cp \"$block/block.json\" \"$block/build/block.json\"; done",
    "start": "wp-scripts start"
  },
  "devDependencies": {
    "@wordpress/scripts": "^27.0.0"
  }
}
```

## Build note

After adding or editing a block, run `npm install && npm run build` from the
plugin root. The build outputs `blocks/<name>/build/`, which is required at
runtime and must be force-included by `plugin-package.sh` even though it is
git-ignored.
